2) Shared formatting helpers (keep it readable)
from datetime import datetime, timezone

def _fmt_ts(ts: Optional[str]) -> str:
    if not ts:
        return "unknown time"
    # Assume ISO-8601; do not re-timezone unless you control it.
    return ts.replace("T", " ").replace(":00-05:00", " (ET)").replace("-05:00", " (ET)")

def _fmt_delta_seconds(sec: Optional[int]) -> str:
    if sec is None:
        return "unknown"
    sec = abs(int(sec))
    m, s = divmod(sec, 60)
    h, m = divmod(m, 60)
    if h:
        return f"{h}h {m}m {s}s"
    if m:
        return f"{m}m {s}s"
    return f"{s}s"

def _fmt_minutes(mins: Optional[float]) -> str:
    if mins is None:
        return "unknown"
    mins = float(mins)
    if mins >= 60:
        h = int(mins // 60)
        m = int(round(mins % 60))
        return f"{h}h {m}m" if m else f"{h}h"
    return f"{int(round(mins))}m"

def _fmt_coord(coord: Optional[dict]) -> str:
    if not coord:
        return "no coordinates"
    lat = coord.get("lat")
    lon = coord.get("lon")
    if lat is None or lon is None:
        return "no coordinates"
    return f"{lat:.6f}, {lon:.6f}"

def _safe_int(x: Any, default: int = 0) -> int:
    try:
        return int(x)
    except Exception:
        return default

def render_location_at_time(e: Evidence) -> List[str]:
    user_id = e.get("user_id", "unknown user")
    w = e.get("window", {}) or {}
    s = e.get("support", {}) or {}
    loc = e.get("location", {}) or {}
    cov = e.get("coverage", {}) or {}

    lines: List[str] = []
    lines.append(f"User: {user_id}")
    lines.append(
        f"Searched {w.get('before_min', '?')} min before and {w.get('after_min', '?')} min after {_fmt_ts(w.get('center_ts'))}."
    )

    lines.append(
        f"Found {_safe_int(s.get('events_in_window'))} phone records in that window; closest record is {_fmt_delta_seconds(s.get('closest_event_delta_sec'))} from the time."
    )

    antenna_id = loc.get("antenna_id")
    coord = loc.get("coord")
    if antenna_id:
        lines.append(f"Closest tower near that time: {antenna_id} ({_fmt_coord(coord)}).")
    else:
        lines.append("No tower/location data available near that time.")

    # Optional: coverage note (still data-first)
    total = cov.get("total_events_in_window")
    with_ant = cov.get("events_with_antenna")
    if total is not None and with_ant is not None:
        lines.append(f"Location data present on {with_ant} of {total} records in the window.")

    # Optional: closest event summary (still factual)
    ce = e.get("closest_event") or {}
    if ce.get("ts"):
        lines.append(_render_event_line(ce))

    return lines

def _render_event_line(ev: Evidence) -> str:
    parts = [
        f"Closest record: {_fmt_ts(ev.get('ts'))}",
        f"{ev.get('interaction','unknown')}",
        f"{ev.get('direction','unknown')}",
    ]
    if ev.get("correspondent_id"):
        parts.append(f"with {ev['correspondent_id']}")
    if ev.get("duration_sec") is not None:
        parts.append(f"duration {ev['duration_sec']}s")
    if ev.get("antenna_id"):
        parts.append(f"tower {ev['antenna_id']}")
    return " | ".join(parts)
3.2 colocation_hit
Expected evidence fields:

users (two IDs)

hit.start_ts / end_ts / duration_min / antenna_id / coord

support.user_events.{user_id}.closest_event_delta_sec / events_in_window

evidence_events (optional lists)

def render_colocation_hit(e: Evidence) -> List[str]:
    users = e.get("users") or []
    u1 = users[0] if len(users) > 0 else "user_1"
    u2 = users[1] if len(users) > 1 else "user_2"

    hit = e.get("hit", {}) or {}
    support = e.get("support", {}) or {}
    per_user = (support.get("user_events", {}) or {})

    lines: List[str] = []
    lines.append(f"Users: {u1} and {u2}")

    antenna = hit.get("antenna_id", "unknown tower")
    start_ts = _fmt_ts(hit.get("start_ts"))
    end_ts = _fmt_ts(hit.get("end_ts"))
    dur = _fmt_minutes(hit.get("duration_min"))
    coord = _fmt_coord(hit.get("coord"))

    lines.append(f"Same tower: {antenna} ({coord}).")
    lines.append(f"Overlap interval: {start_ts} to {end_ts} (duration {dur}).")

    # Per-user support (purely factual)
    for uid in [u1, u2]:
        pu = per_user.get(uid, {}) or {}
        lines.append(
            f"{uid}: closest record is {_fmt_delta_seconds(pu.get('closest_event_delta_sec'))} from overlap; "
            f"{_safe_int(pu.get('events_in_window'))} records in overlap window."
        )

    return lines

    lines.append(f"Tower: {antenna} ({_fmt_coord(hit.get('coord'))}).")
    lines.append(
        f"Interval: {_fmt_ts(hit.get('start_ts'))} to {_fmt_ts(hit.get('end_ts'))} "
        f"(duration {_fmt_minutes(hit.get('duration_min'))})."
    )
    return lines
3.4 synchronized_silence
def render_synchronized_silence(e: Evidence) -> List[str]:
    users = e.get("users") or []
    silence = e.get("silence", {}) or {}
    params = e.get("parameters", {}) or {}

    lines: List[str] = []
    lines.append(f"Devices: {', '.join(users) if users else 'none listed'}")
    lines.append(
        f"No phone records for all listed devices from {_fmt_ts(silence.get('start_ts'))} "
        f"to {_fmt_ts(silence.get('end_ts'))} (duration {_fmt_minutes(silence.get('duration_min'))})."
    )

    # Parameters are still factual and help the investigator understand “what was searched”
    if params:
        lines.append(
            f"Parameters: min overlap {params.get('min_overlap_min','?')}m; gap {params.get('gap_min','?')}m; "
            f"context window {params.get('context_window_min','?')}m."
        )

    # Context counts (keep it simple)
    preceding = e.get("preceding_events", {}) or {}
    following = e.get("following_events", {}) or {}
    for uid in users:
        pre_n = len(preceding.get(uid, []) or [])
        post_n = len(following.get(uid, []) or [])
        lines.append(f"{uid}: {pre_n} records before gap; {post_n} records after gap (within context window).")

    return lines

def render_relationship_summary(e: Evidence) -> List[str]:
    user_id = e.get("user_id", "unknown user")
    contact_id = e.get("contact_id", "unknown contact")
    m = e.get("metrics", {}) or {}
    rank = e.get("rank_fields", {}) or {}

    lines: List[str] = []
    lines.append(f"User: {user_id} | Contact: {contact_id}")
    lines.append(
        f"Total interactions: {m.get('total_interactions',0)} "
        f"({m.get('total_calls',0)} calls, {m.get('total_texts',0)} texts)."
    )
    lines.append(
        f"Direction: outgoing {m.get('outgoing_interactions',0)}, incoming {m.get('incoming_interactions',0)}."
    )
    if m.get("first_contact_ts"):
        lines.append(f"First observed interaction: {_fmt_ts(m.get('first_contact_ts'))}.")
    if m.get("last_contact_ts"):
        lines.append(f"Most recent interaction: {_fmt_ts(m.get('last_contact_ts'))}.")

    # Ranking is allowed if it is deterministic; present it as a computed index, not an interpretation.
    if "relationship_strength_score" in rank:
        lines.append(f"Computed contact-intensity index: {rank.get('relationship_strength_score')} (deterministic).")

    return lines

def render_evidence_compact(e: Evidence) -> str:
    t = e.get("type", "unknown")
    if t == "location_at_time":
        loc = e.get("location", {}) or {}
        s = e.get("support", {}) or {}
        return (
            f"{e.get('user_id','user')} | tower {loc.get('antenna_id','n/a')} | "
            f"closest delta {_fmt_delta_seconds(s.get('closest_event_delta_sec'))}"
        )
    if t == "colocation_hit":
        hit = e.get("hit", {}) or {}
        users = e.get("users") or []
        return (
            f"{'/'.join(users) if users else 'users'} | tower {hit.get('antenna_id','n/a')} | "
            f"{_fmt_ts(hit.get('start_ts'))}–{_fmt_ts(hit.get('end_ts'))}"
        )
    if t == "synchronized_silence":
        sil = e.get("silence", {}) or {}
        return f"Silence | {_fmt_ts(sil.get('start_ts'))}–{_fmt_ts(sil.get('end_ts'))} | {_fmt_minutes(sil.get('duration_min'))}"
    return f"{t}"
5) How this plugs into your existing indicators
Mechanically:

Indicators return evidence dicts with type + data fields.

UI/report export calls render_evidence(e) to display.

If you need tables, use render_evidence_compact(e) as the “summary column.”

This keeps your indicators strictly “pull data, don’t interpret,” while still producing investigator-friendly output.

6) Optional: “Renderer hygiene” rules (recommended)
Never use words: confidence, suspicious, likely, probably, meeting, coordinated.

Allowed words: found, within, closest, same tower, records, window, duration, count.

If something is missing: say “unknown” or “no data present”, not guesses.
